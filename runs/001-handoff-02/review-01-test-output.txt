import importlib.util
import sys

spec = importlib.util.spec_from_file_location("solution", "runs/001-handoff-02/solution.py")
mod = importlib.util.module_from_spec(spec)
spec.loader.exec_module(mod)

CASES = [
    (["C", "A", "C", "B"], ["A", "Z"], ["C", "B"]),
    ([], ["Z"], []),
    (["K", "k"], ["K", "K"], ["k"]),
    (["X"], ["X"], []),
]
failed = 0

    p, d = list(planned), list(done)
    actual = mod.pending_ids(p, d)
    ok = actual == expected and type(actual) is list and p == planned and d == done
